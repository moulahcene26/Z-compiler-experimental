  // Simple integer exponentiation. Prone to overflow.
  int Exp (int a, int b) {
    int i;
    long long Ex = 1; // Use long long for intermediate calculation

    if (b < 0) return 0; // Define result for negative exponent (integer case)
    if (b == 0) return 1;

    for (i = 1; i <= b; i++) {
      // Basic overflow check (not fully reliable)
      if ((a > 0 && Ex > (2147483647 / a)) || (a < 0 && Ex < (-2147483648 / a))) {
           fprintf(stderr, "Warning: Potential integer overflow in Exp(%d, %d).\n", a, b);
           // Return max/min int or specific error?
           return (a > 0) ? 2147483647 : -2147483648;
      }
      Ex = Ex * a;
    }
    // Final check if Ex fits in int range
    if (Ex > 2147483647 || Ex < -2147483648) {
        fprintf(stderr, "Warning: Integer overflow occurred in Exp(%d, %d).\n", a, b);
        return (Ex > 0) ? 2147483647 : -2147483648;
    }
    return (int)Ex;
  }

*** Z ALEACHAINE ***

  // Needs srand(time(NULL)) called once in main.
  // Caller must free the returned string.
  char *Aleachaine ( int N ) {
    int k;
    if (N < 0) return NULL; // Invalid length
    if (N == 0) {
        char *empty_str = malloc(1);
        if (empty_str) empty_str[0] = '\0';
        return empty_str; // Return empty string for N=0
    }

    char *Chaine = malloc(N + 1); // Allocate space for N chars + null terminator
    if (!Chaine) {
        perror("malloc failed in Aleachaine");
        return NULL;
    }

    // Use static const for efficiency
    static const char Chr1[] = "abcdefghijklmnopqrstuvwxyz";
    static const char Chr2[] = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
    const int len1 = sizeof(Chr1) - 1;
    const int len2 = sizeof(Chr2) - 1;

    for (k = 0; k < N; k++) {
      switch ( rand() % 2 ) {
        case 0 : Chaine[k] = Chr1[rand() % len1]; break;
        case 1 : Chaine[k] = Chr2[rand() % len2]; break;
      }
    }
    Chaine[k] = '\0'; // Null-terminate the string

    return Chaine;
  }

  // Renamed in runtime to ZGetCharStr for clarity.
  // Returns a new string containing the character at 1-based index I.
  // Caller must free the returned string.
  char *ZGetCharStr ( const char * Ch , int I ) {
    if (!Ch || I <= 0) return NULL; // Validate input

    int len = strlen(Ch);
    if (I > len) return NULL; // 1-based index out of bounds

    char *s = malloc(2); // Space for character + null terminator
    if (!s) {
        perror("malloc failed in ZGetCharStr");
        return NULL;
    }
    s[0] = Ch[I - 1]; // Convert 1-based index to 0-based
    s[1] = '\0';
    return s;
  }

  /* Internal file tracking list management */

  // Node structure for the linked list
  struct _Noeud {
    FILE * Var_fich ;     // The C file handle
    char * Nom_fich ;     // Allocated copy of the file path/name
    int Sauv_pos;         // Saved position (e.g., during nested opens)
    struct _Noeud *Suiv ; // Pointer to the next node
  };

  typedef struct _Noeud * _Ptr_Noeud;

  // Global head of the list
  _Ptr_Noeud _Pile_ouverts = NULL;

  /* Test if a file path is already tracked */
  _Ptr_Noeud _Ouvert ( const char * Fp) { // Take const char*
    if (!Fp) return NULL;
    _Ptr_Noeud P = _Pile_ouverts;
    while (P != NULL) {
      if (P->Nom_fich && strcmp(P->Nom_fich, Fp) == 0) {
        return P; // Found
      }
      P = P->Suiv;
    }
    return NULL; // Not found
  }

  /* Add a file path and handle to the tracking list */
  void _Empiler_ouvert ( const char *Fp, FILE *Fl) { // Take const char*
    if (!Fp || !Fl) return;

    _Ptr_Noeud P = (_Ptr_Noeud) malloc(sizeof(struct _Noeud));
    if (!P) {
      perror("malloc failed in _Empiler_ouvert");
      return;
    }

    P->Nom_fich = strdup(Fp); // Use strdup to copy the filename
    if (!P->Nom_fich) {
      free(P);
      perror("strdup failed in _Empiler_ouvert");
      return;
    }

    P->Var_fich = Fl;
    // Store current position, default to 0 on error
    P->Sauv_pos = ftell(Fl);
    if (P->Sauv_pos < 0) P->Sauv_pos = 0;

    // Add to head of list
    P->Suiv = _Pile_ouverts;
    _Pile_ouverts = P;
  }

  /* Remove a node by file handle and return its allocated name */
  /* Caller must free the returned string. Returns NULL on failure. */
  char * _Depiler_ouvert ( FILE *Fl) {
    if (_Pile_ouverts == NULL || !Fl) return NULL;

    _Ptr_Noeud P = _Pile_ouverts;
    _Ptr_Noeud Prec = NULL;

    // Find the node
    while (P != NULL && P->Var_fich != Fl ) {
      Prec = P;
      P = P->Suiv;
    }

    if (P == NULL) return NULL; // Handle not found

    // Found node P, make a copy of the name
    char * Fp_copy = strdup(P->Nom_fich);

    // Unlink node P
    if (Prec != NULL) { // Not head
      Prec->Suiv = P->Suiv;
    } else { // Head
      _Pile_ouverts = P->Suiv;
    }

    // Free the node's resources
    free(P->Nom_fich); // Free the filename stored inside the node
    free(P);           // Free the node struct itself

    if (!Fp_copy) {
        perror("strdup failed in _Depiler_ouvert"); // Should not happen if P->Nom_fich was valid
    }
    return Fp_copy;
  }

      ParentNode = NULL;
      P = *Arbre; // Start search from root for each element
      %Typeelem% current_element;

          break; // Stop searching for this duplicate element
        }

  /* Create M-ary Tree from Simple Array (Simplified Insertion) */
  /* %Creer_arm%: Function Name */
  /* %Pointeur%: Node pointer type */
  /* %Typetab%: Source C array type */
  /* N: Number of elements */
  /* Ordre: The 'M' value (passed as param) */
  /* %Creernoeud%, %Aff_degre%, %Aff_infor%, etc. = M-ary op names */
  /* $COMPAR needs comparison logic for %Typeelem% */

  // Creates M-ary tree *Arbre. Uses simplified insertion (new child node on full).
  void %Creer_arm% (%Pointeur% *Arbre, const void* Tab, int N, int Ordre, size_t element_size /*, comparison_func */) {
    bool Trouv1; // Duplicate key found in tree
    bool Trouv2; // Found position within a node (key >= search_key)
    int I, J, Pos; // Loop indices, Position (1-based) in node's Infor/Fils
    int Nt; // Temp for node degree
    %Pointeur% P, Q, R ; // P=current traversal, Q=parent/insert_node, R=new_node

    *Arbre = NULL;
    if (N <= 0 || Ordre <= 1) return; // Basic validation

    // Process Root (Tab[0])
    %Creernoeud% ( &P );
    if (!P) return;
    *Arbre = P;
    %Aff_degre% ( P, 1 );
    %Typeelem% root_element;
    memcpy(&root_element, (const char*)Tab + 0 * element_size, element_size);
    %Aff_infor% ( P, 1, root_element ); // Set first key

    // Process remaining elements Tab[1] to Tab[N-1]
    for (I = 1; I < N; ++I ) {
      P = *Arbre; // Start search from root
      Q = NULL; // Parent/node found
      Trouv1 = false; // Reset duplicate flag
      %Typeelem% current_element;
      memcpy(¤t_element, (const char*)Tab + I * element_size, element_size);

      // Search for insertion node Q and position Pos
      while (P != NULL && !Trouv1) {
        Q = P; // Q is the node we are examining
        Trouv2 = false; // Reset found position flag
        Pos = 1; // 1-based index for Infor/Fils

        // Find first Pos where Infor(Q, Pos) >= current_element
        while (!Trouv2 && (Pos <= %Degre%(Q))) {
          // compare(current_element, %Infor%(Q, Pos)) -> comparison_result
          int comparison_result; // = compare(current_element, %Infor%(Q, Pos));
          if (comparison_result <= 0) { // Found element >= current
            Trouv2 = true;
            if (comparison_result == 0) { // Duplicate
              Trouv1 = true;
              // printf("Duplicate found\n");
            }
          } else {
            Pos++; // Check next key
          }
        } // End while search in node

        if (!Trouv1) { // If not duplicate, decide which child to follow
          // Pos is now the 1-based index of the child to follow
          // (or Degre+1 if current_element > all keys)
          P = %Fils%(Q, Pos);
        }
      } // End while search tree (P == NULL or Trouv1 == true)

      // Insert if not duplicate (Q holds the node for insertion)
      if (!Trouv1 && Q != NULL) { // Ensure Q is valid
        // Check if node Q has space (Degre < Ordre - 1 keys)
        if (%Degre%(Q) < Ordre - 1) {
          // Insert current_element into node Q at position Pos
          Nt = %Degre%(Q);
          // Shift keys and child pointers from Pos onwards right
          for (J = Nt; J >= Pos; --J) {
            %Aff_infor% ( Q, J + 1, %Infor%(Q, J) );
            %Aff_fils% ( Q, J + 2, %Fils%(Q, J + 1) ); // Shift child J+1 to J+2
          }
          // Insert new key and NULL child pointer after it
          %Aff_infor% ( Q, Pos, current_element );
          %Aff_fils% ( Q, Pos + 1, NULL ); // Child pointer after new key
          %Aff_degre% ( Q, Nt + 1 ); // Increment degree
        } else { // Node Q is full - SIMPLIFIED INSERTION (create child)
          %Creernoeud% ( &R ); // New child node
          if (!R) continue; // Malloc check

          %Aff_degre% ( R, 1 ); // New node R has 1 key
          %Aff_infor% ( R, 1, current_element ); // Store key in R
          %Aff_parent% ( R, Q ); // Set R's parent

          // Link R as the child of Q at position Pos
          %Aff_fils% ( Q, Pos, R );
          // NOTE: This does not preserve M-ary/B-tree properties fully.
        } // End if node full/not full
      } // End if (!Trouv1 && Q != NULL)
    } // End for loop
  }
